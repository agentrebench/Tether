cmake_minimum_required(VERSION 3.25)
project(tether_native VERSION 0.4.0 LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(TETHER_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(TETHER_MACOS_APP "Build the Cocoa display adapter" ON)
add_library(tether_runtime STATIC core/tether_runtime.c core/tether_orb.c)
target_include_directories(tether_runtime PUBLIC core)
target_compile_options(tether_runtime PRIVATE -Wall -Wextra -Werror -Wpedantic)
if(TETHER_SANITIZERS)
  target_compile_options(tether_runtime PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(tether_runtime PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  foreach(name runtime orb)
    add_executable(test_${name} tests/test_${name}.c)
    target_link_libraries(test_${name} PRIVATE tether_runtime)
    # Assertions are acceptance checks, including in release builds.
    target_compile_options(test_${name} PRIVATE -Wall -Wextra -Werror -Wpedantic -UNDEBUG)
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 30)
  endforeach()
endif()
if(APPLE AND TETHER_MACOS_APP)
  enable_language(OBJC)
  add_executable(Tether MACOSX_BUNDLE macos/main.m)
  target_compile_options(Tether PRIVATE -fobjc-arc -Wall -Wextra -Werror)
  target_link_libraries(Tether PRIVATE tether_runtime "-framework Cocoa" "-framework WebKit" "-framework QuartzCore")
  set_target_properties(Tether PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/macos/Info.plist")
endif()
